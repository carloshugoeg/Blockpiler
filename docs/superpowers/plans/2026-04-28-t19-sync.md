# T19 SyncManager Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement `frontend/features/sync.js` — the `SyncManager` class that keeps Monaco IDE and Blockly workspace in bidirectional sync via `/api/convert`.

**Architecture:** `SyncManager` wraps the Monaco editor and Blockly workspace references passed at construction. IDE→Blocks changes are debounced 600ms before calling `POST /api/convert?direction=c_to_blocks`; Blocks→IDE changes are immediate. A boolean `_guard` flag prevents re-entrant sync loops.

**Tech Stack:** Vanilla ES modules, Blockly (default import for `serialization.workspaces`), native `fetch`, Vitest + jsdom for tests.

```javascript
// frontend/features/sync.js
import Blockly from 'blockly';

export class SyncManager {
  /**
   * @param {object} monacoEditor  - Monaco editor instance
   * @param {object} blocklyWorkspace - Blockly WorkspaceSvg
   * @param {string} apiBase - URL prefix for API calls (e.g. '' or 'http://localhost:5000')
   */
  constructor(monacoEditor, blocklyWorkspace, apiBase = '') {
    this._editor = monacoEditor;
    this._workspace = blocklyWorkspace;
    this._apiBase = apiBase;
    this._guard = false;
    this._debounceTimer = null;
    this._ideDisposable = null;
    this._blocksListener = null;
  }

  async _syncIdeToBlocks() {
    const source = this._editor.getValue();
    let json;
    try {
      const res = await fetch(`${this._apiBase}/api/convert`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ direction: 'c_to_blocks', source }),
      });
      json = await res.json();
    } catch {
      return;
    }
    if (!json.ok) return;
    this._guard = true;
    try {
      Blockly.serialization.workspaces.load(json.data.workspace, this._workspace);
    } finally {
      this._guard = false;
    }
  }

  async _syncBlocksToIde() {
    const workspaceState = Blockly.serialization.workspaces.save(this._workspace);
    let json;
    try {
      const res = await fetch(`${this._apiBase}/api/convert`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ direction: 'blocks_to_c', workspace: workspaceState }),
      });
      json = await res.json();
    } catch {
      return;
    }
    if (!json.ok) return;
    this._guard = true;
    try {
      this._editor.setValue(json.data.source);
    } finally {
      this._guard = false;
    }
  }

  /** Subscribe to Monaco change events. Returns a disposable. */
  _onIdeChange(handler) {
    return this._editor.onDidChangeModelContent(handler);
  }

  /** Subscribe to Blockly workspace change events. */
  _onBlocksChange(handler) {
    this._workspace.addChangeListener(handler);
  }

  enableSync() {
    this._ideDisposable = this._editor.onDidChangeModelContent(() => {
      if (this._guard) return;
      clearTimeout(this._debounceTimer);
      this._debounceTimer = setTimeout(() => this._syncIdeToBlocks(), 600);
    });

    this._blocksListener = (event) => {
      if (this._guard) return;
      if (event && event.isUiEvent) return;
      this._syncBlocksToIde();
    };
    this._workspace.addChangeListener(this._blocksListener);
  }

  disableSync() {
    if (this._ideDisposable) {
      this._ideDisposable.dispose();
      this._ideDisposable = null;
    }
    if (this._blocksListener) {
      this._workspace.removeChangeListener(this._blocksListener);
      this._blocksListener = null;
    }
    clearTimeout(this._debounceTimer);
    this._debounceTimer = null;
  }
}
```
